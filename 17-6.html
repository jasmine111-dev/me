// Promise 化 setTimeout
function delay(ms, name) {
    return new Promise(resolve => {
        setTimeout(() => resolve(name), ms);
    });
}

async function makeDumplings() {
    // ① Promise.race：线上 + 线下，谁先完用谁
    let result = await Promise.race([
        delay(5000, "线上买原料"),
        delay(6000, "线下买原料")
    ]);
    console.log("胜者是：" + result);

    // ② 取原料 1s（串行）
    await delay(1000, "取原料");

    // ③④ Promise.all：和面 + 和馅，都要完
    await Promise.all([
        delay(3000, "和面"),
        delay(2000, "和馅")
    ]);

    // ⑤ 包饺子 4s（串行）
    await delay(4000, "包饺子");

    // ⑥ 煮饺子 3s（串行）— 新增步骤
    await delay(3000, "煮饺子");

    // ⑦ 吃饺子 2s（串行）
    await delay(2000, "吃饺子");
}

// ⚡ 两种 Promise 静态方法对比：
// Promise.race  —— 竞速，第一个完成/失败就返回（取最快者）
// Promise.all   —— 结伴，全部完成才返回（取最慢者），一个失败全失败
