// 用 Promise 包装 setTimeout，让它支持 await
function delay(ms, name) {
    return new Promise(resolve => {
        setTimeout(() => resolve(name), ms);
    });
}

// 主流程：async 函数 + await 串行 + Promise.all 并发
async function makeDumplings() {
    let t0 = performance.now();

    // ① 串行：买原料 6s
    await delay(6000, "买原料");

    // ② 串行：取原料 1s
    await delay(1000, "取原料");

    // ③④ 并发：和面 3s + 和馅 2s —— 同时开始，取慢者 3s
    await Promise.all([
        delay(3000, "和面"),
        delay(2000, "和馅")
    ]);

    // ⑤ 串行：包饺子 4s
    await delay(4000, "包饺子");

    // ⑥ 串行：吃饺子 2s
    await delay(2000, "吃饺子");

    return ((performance.now() - t0) | 0);
}

// ⚡ 对比：如果全部串行（没有 Promise.all）
// 总耗时 = 6 + 1 + 3 + 2 + 4 + 2 = 18s
// 用 Promise.all 并发和面和馅后
// 总耗时 = 6 + 1 + max(3,2) + 4 + 2 = 16s —— 省了 2s
